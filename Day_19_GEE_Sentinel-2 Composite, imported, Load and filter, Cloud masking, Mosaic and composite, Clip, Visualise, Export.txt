Sentinel-2 Composite, imported, Load and filter, Cloud masking, Mosaic and composite, Clip, Visualise, Export

// ===== Delhi Sentinel-2 Composite: full script =====

// 1. Delhi boundary (imported from Assets as 'table')
var delhi = table;
Map.centerObject(delhi, 10);

// 2. Load and filter Sentinel-2
var s2 = ee.ImageCollection('COPERNICUS/S2_SR_HARMONIZED')
  .filterBounds(delhi)
  .filterDate('2026-03-01', '2026-04-30')
  .filter(ee.Filter.lt('CLOUDY_PIXEL_PERCENTAGE', 20));
print('Number of images found:', s2.size());

// 3. Cloud masking
function maskClouds(image) {
  var qa = image.select('QA60');
  var clear = qa.bitwiseAnd(1 << 10).eq(0)
           .and(qa.bitwiseAnd(1 << 11).eq(0));
  return image.updateMask(clear).divide(10000)
              .copyProperties(image, ['system:time_start']);
}
var s2Clean = s2.map(maskClouds);

// 4. Mosaic and composite
var mosaic    = s2Clean.mosaic();
var composite = s2Clean.median();

// 5. Clip to Delhi
var mosaicDelhi    = mosaic.clip(delhi);
var compositeDelhi = composite.clip(delhi);

// 6. Visualise
var trueColour  = {bands: ['B4', 'B3', 'B2'], min: 0, max: 0.3};
var falseColour = {bands: ['B8', 'B4', 'B3'], min: 0, max: 0.4};
Map.addLayer(mosaicDelhi, trueColour, 'Mosaic - true colour');
Map.addLayer(compositeDelhi, trueColour, 'Composite - true colour');
Map.addLayer(compositeDelhi, falseColour, 'Composite - false colour');
Map.addLayer(delhi.style({color: 'red', fillColor: '00000000'}), {}, 'Delhi boundary');

// 7. Export
Export.image.toDrive({
  image: compositeDelhi.select(['B2', 'B3', 'B4', 'B8']).toFloat(),
  description: 'Delhi_S2_Composite',
  folder: 'GEE_Exports',
  fileNamePrefix: 'Delhi_S2_Composite',
  region: delhi.geometry(),
  scale: 10,
  crs: 'EPSG:32643',
  maxPixels: 1e13
});
